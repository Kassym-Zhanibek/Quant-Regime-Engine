```py
def get_yesterdays_decayed_sentiment():
    conn = get_connection()
    yesterday_str = (datetime.now(timezone.utc) - timedelta(days=1)).strftime('%Y-%m-%d 00:00:00')
    today_str = datetime.now(timezone.utc).strftime('%Y-%m-%d 00:00:00')

    cursor = conn.cursor()
    cursor.execute("""
        SELECT finbert_score, qwen_score, finbert_neu, category 
        FROM news WHERE published_at >= ? AND published_at < ?
    """, (yesterday_str, today_str))
    rows = cursor.fetchall()
    conn.close()

    if not rows:
        return 0.0, 0.0

    fb_num, qwen_num, den = 0.0, 0.0, 0.0
    for r in rows:
        w_cat = 1.5 if r["category"] == "DIRECT_GOLD" else 1.0
        w_impact = w_cat * (1.0 - (r["finbert_neu"] or 0.0))
        fb_num += (r["finbert_score"] or 0.0) * w_impact
        qwen_num += (r["qwen_score"] or 0.0) * w_impact
        den += w_impact

    avg_fb = fb_num / den if den > 0 else 0.0
    avg_qwen = qwen_num / den if den > 0 else 0.0
    decay_factor = math.exp(-0.693)
    return avg_fb * decay_factor, avg_qwen * decay_factor

def _render_page(name: str, active: str) -> HTMLResponse:
    """Читает шаблон и подставляет в <!--NAV--> общую навигацию (_nav.html), подсвечивая активную вкладку."""
    path = os.path.join(TEMPLATES_DIR, name)
    if not os.path.exists(path):
        return HTMLResponse(f"<h1>{name} не найден в app/templates</h1>", status_code=404)
    with open(path, "r", encoding="utf-8") as f:
        html = f.read()
    nav_path = os.path.join(TEMPLATES_DIR, "_nav.html")
    if os.path.exists(nav_path):
        with open(nav_path, "r", encoding="utf-8") as f:
            nav = (f.read().replace("{{GOLD_CLS}}", NAV_ON if active == "gold" else NAV_OFF)
                           .replace("{{STOCKS_CLS}}", NAV_ON if active == "stocks" else NAV_OFF))
        html = html.replace("<!--NAV-->", nav)
    return HTMLResponse(html)

@app.get("/api/v1/sentiment/gold", response_model=FullNlpPipelineResponse)
async def get_gold_sentiment():
    macro_data = await asyncio.to_thread(fetch_macro_indicators)

    cutoff_time = datetime.now(timezone.utc) - timedelta(hours=24)

    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
        SELECT finbert_score, qwen_score, finbert_neu, category, raw_score 
        FROM news WHERE published_at >= ?
    """, (cutoff_time.strftime('%Y-%m-%d %H:%M:%S'),))
    news_rows = cursor.fetchall()
    conn.close()

    if not news_rows:
        avg_fb, avg_qwen = await asyncio.to_thread(get_yesterdays_decayed_sentiment)
        active_count = 0
        latest_score = avg_fb
    else:
        fb_num, qwen_num, den = 0.0, 0.0, 0.0
        for r in news_rows:
            w_cat = 1.5 if r["category"] == "DIRECT_GOLD" else 1.0
            w_impact = w_cat * (1.0 - (r["finbert_neu"] or 0.0))
            fb_num += (r["finbert_score"] or 0.0) * w_impact
            qwen_num += (r["qwen_score"] or 0.0) * w_impact
            den += w_impact

        avg_fb = fb_num / den if den > 0 else 0.0
        avg_qwen = qwen_num / den if den > 0 else 0.0
        active_count = len(news_rows)
        latest_score = news_rows[0]["raw_score"]

    dxy_delta = macro_data["dxy"]["change_pct"]
    us10y_delta_bps = macro_data["us10y_yields"]["change_bps"]

    if rf_model:
        features = pd.DataFrame({
            'finbert_score': [avg_fb],
            'qwen_score': [avg_qwen],
            'dxy_change': [dxy_delta],
            'fed_change': [us10y_delta_bps]
        })
        probs = rf_model.predict_proba(features)[0]
        prob_down, prob_up = probs[0], probs[1]

        composite_bias = round(prob_up - prob_down, 4)
        confidence = round(abs(prob_up - 0.5) / 0.5, 2)
    else:
        nlp_score = math.tanh(((avg_fb * 0.3) + (avg_qwen * 0.7)) / 2.0)
        composite_bias = round(nlp_score * 0.7 + macro_data["macro_score"] * 0.3, 4)
        confidence = round(abs(composite_bias), 2)

    label = "BULLISH" if composite_bias >= 0.20 else "BEARISH" if composite_bias <= -0.20 else "NEUTRAL"

    save_sentiment_snapshot(composite_bias, round(latest_score, 4), macro_data["macro_score"])

    return {
        "asset": "GC (Gold)",
        "market_bias_score": composite_bias,
        "latest_sentiment_score": round(latest_score, 4),
        "bias_label": label,
        "confidence": confidence,
        "active_news_count": active_count,
        "dxy": macro_data["dxy"],
        "us10y_yields": macro_data["us10y_yields"],
        "macro_score": macro_data["macro_score"],
        "last_updated": datetime.now(timezone.utc).isoformat()
    }
```

```py
import yfinance as yf


def fetch_macro_indicators() -> dict:
    """
    Скачивает макро-данные и считает изменения:
    DXY - относительное изменение (в долях), US10Y - изменение в базисных пунктах (bps).
    """
    try:
        dxy_data = yf.Ticker("DX-Y.NYB").history(period="7d", interval="1d")
        tnx_data = yf.Ticker("^TNX").history(period="7d", interval="1d")

        if len(dxy_data) < 2 or len(tnx_data) < 2:
            return get_fallback_data()

        dxy_prev_close = dxy_data['Close'].iloc[-2]
        dxy_live = dxy_data['Close'].iloc[-1]

        tnx_prev_close = tnx_data['Close'].iloc[-2]
        tnx_live = tnx_data['Close'].iloc[-1]

        # DXY delta в долях (например, 0.0015 для +0.15%)
        dxy_delta = (dxy_live - dxy_prev_close) / dxy_prev_close
        # Доходность 10Y в базисных пунктах (bps)
        us10y_delta_bps = (tnx_live - tnx_prev_close) * 100.0

        macro_score = max(-1.0, min(1.0, round(-(dxy_delta * 10.0 + (us10y_delta_bps / 100.0)), 4)))

        return {
            "dxy": {
                "price": round(float(dxy_live), 2),
                "change_pct": round(float(dxy_delta), 4)
            },
            "us10y_yields": {
                "yield_pct": round(float(tnx_live), 2),
                "change_bps": round(float(us10y_delta_bps), 1)
            },
            "macro_score": macro_score
        }
    except Exception as e:
        print(f"Ошибка сбора макро-данных yfinance: {e}")
        return get_fallback_data()

        "us10y_yields": {"yield_pct": 4.2, "change_bps": 0.0},
        "macro_score": 0.0

import threading
import time

_MARKETS_LOCK = threading.Lock()
_MARKETS_CACHE: dict = {"ts": 0.0, "data": None}


def _quote(ticker: str, kind: str, name: str) -> dict:
    """Последняя цена и изменение к предыдущему закрытию. change_pct — в ПРОЦЕНТАХ (0.15 = +0.15%), change_bps — в б.п."""
    out = {"name": name, "ticker": ticker, "kind": kind, "price": None, "change_pct": None, "change_bps": None}
    try:
        hist = yf.Ticker(ticker).history(period="7d", interval="1d")["Close"].dropna()
        if len(hist) < 2:
            return out
        prev, last = float(hist.iloc[-2]), float(hist.iloc[-1])
        out["price"] = round(last, 3 if kind == "yield" else 2)
        if kind == "yield":
            out["change_bps"] = round((last - prev) * 100.0, 1)      # ^TNX котируется в процентах -> б.п.
        else:
            out["change_pct"] = round((last - prev) / prev * 100.0, 2)
    except Exception as e:
        print(f"Ошибка котировки {ticker}: {e}")
    return out


def get_markets() -> dict:

from concurrent.futures import ThreadPoolExecutor
from datetime import date, datetime

_STOCKS_LOCK = threading.Lock()
_STOCKS_CACHE: dict = {"ts": 0.0, "data": None}
_EARN_CACHE: dict = {}                       # ticker -> (время запроса, 'YYYY-MM-DD' | None)
EARNINGS_TTL = 6 * 3600                      # дату отчёта меняют редко
EARNINGS_MISS_TTL = 30 * 60                  # не нашли — пробуем чаще


def _stock_quote(ticker: str, name: str) -> dict:
    """Цена, изменение к пред. закрытию (%) и за 5 сессий (%). При ошибке поля остаются None."""
    out = {"name": name, "ticker": ticker, "kind": "price", "price": None, "change_pct": None,
           "change_5d_pct": None, "change_bps": None}
    try:
        hist = yf.Ticker(ticker).history(period="1mo", interval="1d")["Close"].dropna()
        if len(hist) >= 2:
            last, prev = float(hist.iloc[-1]), float(hist.iloc[-2])
            out["price"] = round(last, 2)
            out["change_pct"] = round((last - prev) / prev * 100.0, 2)
            if len(hist) >= 6:
                base = float(hist.iloc[-6])
                out["change_5d_pct"] = round((last - base) / base * 100.0, 2)
    except Exception as e:
        print(f"Ошибка котировки {ticker}: {e}")
    return out


def _to_date(x):
    if isinstance(x, datetime):
        return x.date()
    if isinstance(x, date):
        return x
    try:
        return date.fromisoformat(str(x)[:10])
    except ValueError:
        return None


def _next_earnings(ticker: str) -> str | None:
    """Ближайшая дата отчётности ('YYYY-MM-DD') по данным Yahoo; None, если не удалось получить (yfinance капризен)."""
    now = time.time()
    hit = _EARN_CACHE.get(ticker)
    if hit and now - hit[0] < (EARNINGS_TTL if hit[1] else EARNINGS_MISS_TTL):
        return hit[1]
    value = None
    try:
        cal = yf.Ticker(ticker).calendar
        raw = cal.get("Earnings Date") if isinstance(cal, dict) else None
        if raw is None and cal is not None and hasattr(cal, "loc"):        # старые версии yfinance: DataFrame
            raw = cal.loc["Earnings Date"].tolist()
        if raw is not None and not isinstance(raw, (list, tuple)):
            raw = [raw]
        today = date.today()
        future = sorted(d for d in (_to_date(x) for x in (raw or [])) if d is not None and d >= today)
        value = future[0].isoformat() if future else None
    except Exception as e:
        print(f"Не получена дата отчёта {ticker}: {e}")
    _EARN_CACHE[ticker] = (now, value)
    return value


def get_stock_quotes() -> dict:
    """
    {key: {...котировка, 'markets': [...], 'next_earnings', 'days_to_earnings'}} для «Рынка США» и тикеров.
    Для 'spx' в 'markets' лежат S&P 500 / Nasdaq / VIX, для акции — её собственная котировка. Кэш MARKET_CACHE_SECONDS.
    """
    from app import config
    from app.categories import CATEGORIES, STOCK_ORDER
    with _STOCKS_LOCK:
        if _STOCKS_CACHE["data"] is not None and time.time() - _STOCKS_CACHE["ts"] < config.MARKET_CACHE_SECONDS:
            return _STOCKS_CACHE["data"]

        def one(key: str) -> dict:
            cat = CATEGORIES[key]
            if cat["markets"]:                                  # «Рынок США»: индексы
                return {"key": key, "markets": [_quote(m["ticker"], m["kind"], m["name"]) for m in cat["markets"]]}
            q = _stock_quote(cat["ticker"], cat["name"])
            earn = _next_earnings(cat["ticker"])
            days = (date.fromisoformat(earn) - date.today()).days if earn else None
            return {"key": key, **{k: q[k] for k in ("price", "change_pct", "change_5d_pct")},
                    "next_earnings": earn, "days_to_earnings": days, "markets": [q]}

        with ThreadPoolExecutor(max_workers=6) as ex:
            data = dict(zip(STOCK_ORDER, ex.map(one, STOCK_ORDER)))
        _STOCKS_CACHE.update(ts=time.time(), data=data)
        return data
```

```py
"""
Загрузка RSS-лент для общего конвейера (источники — app/sources.py, kind = "rss").

Каждая запись ленты превращается в «пост» так же, как сообщение Telegram, и дальше идёт по одному и тому же
пути: фильтр Qwen -> FinBERT -> Qwen U/D/N по категориям. Прежний RSS-конвейер (rss_parser.py -> таблица news)
этот модуль не трогает.
"""
import asyncio
import calendar
import html
import logging
import re
from datetime import datetime, timedelta, timezone

import feedparser
import httpx

from app import config
from app.scoring.text_clean import ANY_URL, SPACES
from app.storage.posts_db import utc_str

logger = logging.getLogger(__name__)

# Заголовки обычного браузера: часть сайтов отдаёт 401/403 «ботам»
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
    "Accept": "application/rss+xml,application/xml;q=0.9,text/xml;q=0.8,*/*;q=0.7",
    "Accept-Language": "en-US,en;q=0.5",
}
TAGS = re.compile(r"<[^>]+>")


def strip_html(s: str | None) -> str:
    return SPACES.sub(" ", html.unescape(TAGS.sub(" ", s or ""))).strip()


def build_text(title: str, summary: str) -> str:
    """Заголовок + короткое описание (если оно не повторяет заголовок)."""
    title, summary = title.strip(), summary.strip()
    if summary.casefold().startswith(title.casefold()):        # описание повторяет заголовок — оставляем только хвост
        summary = summary[len(title):].lstrip(" .:-—")
    if len(summary) > config.RSS_MAX_SUMMARY_CHARS:
        summary = summary[: config.RSS_MAX_SUMMARY_CHARS].rsplit(" ", 1)[0] + "…"
    if not summary or summary.casefold() == title.casefold():
        return title
    sep = "" if title.endswith((".", "!", "?", ":")) else "."
    return f"{title}{sep} {summary}"


def parse_feed_text(src: dict, xml_text: str, now: datetime | None = None) -> list[dict]:
    """Чистая функция: текст ленты -> список постов (dict'ы для posts_db.insert_posts)."""
    now = now or datetime.now(timezone.utc)
    oldest = now - timedelta(hours=config.RSS_MAX_AGE_HOURS)
    parsed = feedparser.parse(xml_text)
    key = f"rss:{src['id']}"
    rows = []
    for e in parsed.entries:
        title = strip_html(e.get("title"))
        if not title:
            continue
        summary = "" if src.get("title_only") else strip_html(e.get("summary") or e.get("description"))
        ts = e.get("published_parsed") or e.get("updated_parsed")
        dt = datetime.fromtimestamp(calendar.timegm(ts), tz=timezone.utc) if ts else now
        if dt < oldest:
            continue
        if dt > now + timedelta(hours=1):          # часовые пояса в лентах иногда врут
            dt = now
        text = build_text(title, summary)
        clean = SPACES.sub(" ", ANY_URL.sub("", text)).strip()
        post_id = e.get("id") or e.get("link") or f"{title}|{utc_str(dt)}"
        rows.append({"source": key, "post_id": str(post_id)[:500], "kind": "rss", "posted_at": utc_str(dt),
                     "title": title, "text": text, "clean_text": clean, "hashtags": None, "url": e.get("link")})
    return rows


async def fetch_feed(http: httpx.AsyncClient, src: dict) -> tuple[list[dict], str | None]:
    """(посты, ошибка). Ошибка одной ленты не мешает остальным."""
    try:
        r = await http.get(src["url"], headers=HEADERS, follow_redirects=True, timeout=config.RSS_TIMEOUT)
        r.raise_for_status()
        rows = await asyncio.to_thread(parse_feed_text, src, r.text)
        return rows, None
    except Exception as e:
        logger.warning("RSS %s: %s", src["id"], e)
        return [], f"{type(e).__name__}: {e}"[:200]


async def fetch_all(http: httpx.AsyncClient, srcs: list[dict]) -> tuple[list[dict], dict]:
    """Скачивает все ленты (по 4 одновременно). Возвращает (все посты, {id: {'entries': n, 'error': str|None}})."""
    sem = asyncio.Semaphore(4)

    async def one(src):
        async with sem:
            return src, *await fetch_feed(http, src)

    results = await asyncio.gather(*(one(s) for s in srcs))
    rows, report = [], {}
    for src, r, err in results:
        rows += r
        report[src["id"]] = {"entries": len(r), "error": err}
    return rows, report
```

```py
import threading

import torch
from transformers import AutoTokenizer, AutoModelForSequenceClassification


class FinBERTGatekeeper:
    def __init__(self, model_name: str = "ProsusAI/finbert"):
        # GPU, если есть (RTX 4090 ускоряет скоринг Telegram-постов в разы), иначе CPU
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_name).to(self.device)
        self.model.eval()
        # RSS-воркер и Telegram-воркер работают в разных потоках с ОДНОЙ моделью — инференс под замком
        self._lock = threading.Lock()

    def probabilities(self, text: str) -> tuple[float, float, float]:
        """Сырые вероятности (positive, negative, neutral) без округления."""
        inputs = self.tokenizer(
            text,
            return_tensors="pt",
            padding=True,
            truncation=True,
            max_length=512
        ).to(self.device)

        with self._lock, torch.no_grad():
            outputs = self.model(**inputs)
            probs = torch.nn.functional.softmax(outputs.logits, dim=-1)[0].cpu()

        # В ProsusAI/finbert: 0 -> positive, 1 -> negative, 2 -> neutral
        return probs[0].item(), probs[1].item(), probs[2].item()

    def analyze_text(self, text: str, neutral_threshold: float = 0.95) -> dict:
        """
        1 этап конвейера:
        Возвращает 3 вероятности (positive, negative, neutral),
        вычисляет базовый балл FinBERT (pos - neg) и выполняет фильтрацию шума.
        """
        p_positive, p_negative, p_neutral = self.probabilities(text)

        # Первичный балл FinBERT (-1.0 ... +1.0)
        finbert_score = p_positive - p_negative

        # Базовая отсечка информационного шума
        is_relevant = p_neutral < neutral_threshold

        return {
            "p_positive": round(p_positive, 4),
            "p_negative": round(p_negative, 4),
            "p_neutral": round(p_neutral, 4),
            "finbert_score": round(finbert_score, 4),
            "is_relevant": is_relevant
        }


_instance: FinBERTGatekeeper | None = None
_instance_lock = threading.Lock()


def get_finbert() -> FinBERTGatekeeper:
    """Единый экземпляр FinBERT на всё приложение (RSS-конвейер и Telegram-конвейер делят одну модель)."""
    global _instance
    with _instance_lock:
        if _instance is None:
            _instance = FinBERTGatekeeper()
        return _instance
```

```py
"""
Агрегация постовых скоров в итоговый скор «общего новостного фона» на момент as-of.
Математика один-в-один из score_live.py (чистые функции, без обращений к базе).

Для каждого полупериода H (часы) и ОТДЕЛЬНО для FinBERT и Qwen:

    w_i    = 0.5 ** (age_i / H)         затухание: пост возраста H часов весит вдвое меньше свежего
    W      = Σ w_i                      «свежая масса» новостей
    avg    = Σ w_i·s_i / W              взвешенное среднее, [-1, 1]
    shrunk = Σ w_i·s_i / (W + k)        ИТОГОВЫЙ скор: то же, но с «якорем» из k псевдо-нейтральных постов
    z      = Σ w_i·(s_i − μ) / σ        затухающий импульс в единицах σ (μ, σ — из истории скоров)

Заглядывания в будущее нет: берутся только посты с posted_at <= as-of.
"""
from datetime import datetime, timedelta, timezone

import numpy as np

TS_FMT = "%Y-%m-%d %H:%M:%S"


def hl_tag(hl: float) -> str:
    return f"h{hl:g}".replace(".", "_")


def to_arrays(rows: list[tuple]):
    """rows: (posted_at 'YYYY-MM-DD HH:MM:SS' UTC, finbert_score|None, qwen_score|None) -> t, fb, qw (numpy)."""
    t = np.array([datetime.strptime(r[0], TS_FMT).replace(tzinfo=timezone.utc).timestamp() for r in rows], dtype=float)
    fb = np.array([np.nan if r[1] is None else r[1] for r in rows], dtype=float)
    qw = np.array([np.nan if r[2] is None else r[2] for r in rows], dtype=float)
    return t, fb, qw


def decayed(t, s, asof_ts, hl, mu, sigma, k) -> dict:
    m = (t <= asof_ts) & ~np.isnan(s)
    if not m.any():
        return {"avg": None, "shrunk": 0.0, "z": 0.0, "w": 0.0}
    age_h = (asof_ts - t[m]) / 3600.0
    w = 0.5 ** (age_h / hl)
    W = float(w.sum())
    ssum = float((w * s[m]).sum())
    return {
        "avg": ssum / W,
        "shrunk": ssum / (W + k),
        "z": float((w * (s[m] - mu)).sum() / sigma) if sigma > 0 else 0.0,
        "w": W,
    }


def build_row(asof: datetime, t, fb, qw, stats: dict, half_lives, k) -> dict:
    """Плоская строка (как в таблице sentiment_asof_live): asof, finbert_n, finbert_avg_h24, finbert_shrunk_h24, ..."""
    ts = asof.timestamp()
    row = {"asof": asof.strftime(TS_FMT)}
    for name, arr in (("finbert", fb), ("qwen", qw)):
        mu, sigma, _ = stats[name]
        m = (t <= ts) & ~np.isnan(arr)
        row[f"{name}_n"] = int(m.sum())
        row[f"{name}_raw_mean"] = float(arr[m].mean()) if m.any() else None      # без затухания, для сравнения
        for hl in half_lives:
            f = decayed(t, arr, ts, hl, mu, sigma, k)
            tag = hl_tag(hl)
            row[f"{name}_avg_{tag}"] = f["avg"]
            row[f"{name}_shrunk_{tag}"] = f["shrunk"]
            row[f"{name}_z_{tag}"] = f["z"]
            row[f"{name}_w_{tag}"] = f["w"]
    return row


def build_grid(asof: datetime, hours: int) -> list[datetime]:
    """Целые часы за последние `hours` часов + сама точка as-of."""
    start = (asof - timedelta(hours=hours)).replace(minute=0, second=0, microsecond=0)
    grid, h = [], start
    while h < asof:
        grid.append(h)
        h += timedelta(hours=1)
    grid.append(asof)
    return grid


def row_to_table(row: dict, half_lives) -> list[dict]:
    """Плоская строка -> [{hl, finbert:{shrunk,avg,z,w}, qwen:{...}}] для API / панели."""
    out = []
    for hl in half_lives:
        tag = hl_tag(hl)
        item = {"hl": hl}
        for name in ("finbert", "qwen"):
            item[name] = {
                "shrunk": row[f"{name}_shrunk_{tag}"],
                "avg": row[f"{name}_avg_{tag}"],
                "z": row[f"{name}_z_{tag}"],
                "w": row[f"{name}_w_{tag}"],
            }
        out.append(item)
    return out
```

```py
"""
Автоматический конвейер «Telegram + RSS -> FinBERT + Qwen -> скоры по категориям».

Один цикл (воркер раз в TG_POLL_SECONDS и кнопка «Докачать посты» на панели):

  1. fetch Telegram   новые посты каналов из app/sources.py (от последнего сохранённого message_id)
  2. fetch RSS        новые записи лент (не чаще, чем раз в RSS_POLL_SECONDS)
  3. routing/dedupe   в какие категории попал пост (источник + ключевые слова), точные дубли внутри суток
  4. judge            Qwen-фильтр важности (1/0): отдельный для золота/макро (исходный промпт) и для акций США
  5. score            FinBERT — один на новость; Qwen U/D/N — по разу на каждую категорию (свой промпт);
                      FinBERT и Qwen пишутся независимо: если Qwen упал, FinBERT-скор сохраняется
  6. aggregate        затухание, shrunk / avg / z / W по каждой категории; журнал в sentiment_asof

Любая стадия может упасть — остальные всё равно выполняются; причина попадает в pipeline.status.
"""
import asyncio
import logging
import random
import time
from datetime import datetime, timedelta, timezone

import httpx

from app import config, sources
from app.categories import ALL_ORDER, STOCK_ORDER
from app.collector import rss_ingest
from app.scoring import scorer, service
from app.scoring.text_clean import clean_telegram_message, cyrillic_share
from app.storage import posts_db

logger = logging.getLogger(__name__)

BATCH = 20          # как часто сбрасывать накопленные скоры в базу


class Pipeline:
    def __init__(self):
        self._lock: asyncio.Lock | None = None       # создаётся лениво, уже внутри event loop
        self._client = None
        self._legacy_imported = False
        self._last_rss = 0.0
        self._routed_full = False         # полный пересчёт маршрутизации (за весь период) — один раз после запуска
        self.status: dict = {
            "enabled": config.TG_ENABLED,
            "telegram": "idle",          # ok | disabled | no_credentials | telethon_missing | needs_login | error
            "stage": "idle",             # idle | fetch | rss | routing | judge | score | aggregate
            "running": False,
            "last_cycle_at": None,
            "last_cycle_seconds": None,
            "sources": {},               # id -> {"kind", "new", "error"}
            "errors": [],
        }

    # ------------------------------------------------------------------ служебное

    @property
    def lock(self) -> asyncio.Lock:
        if self._lock is None:
            self._lock = asyncio.Lock()
        return self._lock

    def _stage(self, name: str) -> None:
        self.status["stage"] = name

    def _err(self, where: str, e: Exception | str) -> None:
        msg = f"{where}: {e}"
        logger.error(msg)
        self.status["errors"].append(msg)

    # ------------------------------------------------------------------ Telegram

    async def _get_client(self):
        """Подключённый и авторизованный клиент Telethon либо None (причина — в status['telegram'])."""
        if not config.TG_ENABLED:
            self.status["telegram"] = "disabled"
            return None
        if not (config.TG_API_ID and config.TG_API_HASH):
            self.status["telegram"] = "no_credentials"
            return None
        try:
            from telethon import TelegramClient
        except ImportError:
            self.status["telegram"] = "telethon_missing"
            return None

        try:
            if self._client is None:
                self._client = TelegramClient(config.TG_SESSION, int(config.TG_API_ID), config.TG_API_HASH)
            if not self._client.is_connected():
                await self._client.connect()
            if not await self._client.is_user_authorized():
                self.status["telegram"] = "needs_login"
                return None
        except Exception as e:
            self.status["telegram"] = "error"
            self._err("telegram connect", e)
            return None
        self.status["telegram"] = "ok"
        return self._client

    async def _fetch_tg(self) -> None:
        self._stage("fetch")
        client = await self._get_client()
        if client is None:
            return
        for src in sources.tg_sources():
            channel = src["id"]
            try:
                last_id = await asyncio.to_thread(posts_db.last_tg_message_id, channel)
                if last_id:
                    # только новее последнего сохранённого, от старых к новым
                    it = client.iter_messages(channel, min_id=last_id, reverse=True)
                else:
                    start = datetime.now(timezone.utc) - timedelta(hours=config.TG_BACKFILL_HOURS)
                    it = client.iter_messages(channel, offset_date=start - timedelta(seconds=1), reverse=True)

                rows = []
                async for msg in it:
                    if not msg.text:                       # картинки/стикеры без текста
                        continue
                    clean, tags = clean_telegram_message(channel, msg.text)
                    rows.append({"source": channel, "post_id": msg.id, "kind": "tg",
                                 "posted_at": posts_db.utc_str(msg.date), "title": None, "text": msg.text,
                                 "clean_text": clean, "hashtags": tags, "url": f"https://t.me/{channel}/{msg.id}"})
                    if len(rows) >= config.TG_MAX_PER_CYCLE:
                        break
                n = await asyncio.to_thread(posts_db.insert_posts, rows)
                self.status["sources"][channel] = {"kind": "tg", "new": n, "error": None}
            except Exception as e:
                self.status["sources"][channel] = {"kind": "tg", "new": 0, "error": f"{type(e).__name__}: {e}"[:200]}
                self._err(f"fetch @{channel}", e)

    # ------------------------------------------------------------------ RSS

    async def _fetch_rss(self, http: httpx.AsyncClient, force: bool) -> None:
        if not force and time.monotonic() - self._last_rss < config.RSS_POLL_SECONDS:
            return
        self._stage("rss")
        rows, report = await rss_ingest.fetch_all(http, sources.rss_sources())
        self._last_rss = time.monotonic()
        n = await asyncio.to_thread(posts_db.insert_posts, rows)
        # «новые» по каждой ленте точно не посчитать без лишнего запроса, поэтому показываем число записей в ленте
        for sid, rep in report.items():
            self.status["sources"][sid] = {"kind": "rss", "new": rep["entries"], "error": rep["error"]}
        failed = [sid for sid, rep in report.items() if rep["error"]]
        if failed:
            self._err("rss", f"не открылись ленты: {', '.join(failed)} (см. scripts/check_sources.py)")
        logger.info("RSS: записей в лентах %s, новых в базе %s", len(rows), n)

    # ------------------------------------------------------------------ Qwen-фильтр

    async def _judge(self, http: httpx.AsyncClient) -> None:
        """Золото/макро и акции проверяются РАЗНЫМИ промптами и пишут вердикты в разные колонки. Золото — первым и без лимита."""
        self._stage("judge")
        for group in ("gold", "stocks"):
            if group == "stocks" and not STOCK_ORDER:
                continue
            limit = config.STOCK_JUDGE_PER_CYCLE if group == "stocks" else None
            if not await self._judge_group(http, group, limit):
                return                                           # Ollama не отвечает — остальное на следующем цикле

    async def _judge_group(self, http: httpx.AsyncClient, group: str, limit: int | None) -> bool:
        rows = await asyncio.to_thread(posts_db.pending_judge, group, limit)
        total = len(rows)

                    self._err("judge", f"{consec} неудач подряд — Ollama не отвечает, остальное на следующем цикле")
                    return False
                continue
            consec = 0
            await asyncio.to_thread(posts_db.set_verdict, source, post_id, verdict, group)

            preview = (text or "").replace("\n", " ")[:60]
            logger.info("[%s %d/%d] [%s] %s | \"%s...\"", tag, idx, total, source, status_str, preview)

                batch = []

            total = len(rows)

            batch = []

                res = await scorer.qwen_udn(http, text, category)
                if res is None:
                    consec += 1
                    if consec >= config.MAX_CONSEC_QWEN_FAIL:
                        self._err("qwen score", f"{consec} неудач подряд — Ollama не отвечает, остальное на следующем цикле")
                        if batch:
                            await asyncio.to_thread(posts_db.save_qwen, category, batch)
                        return
                    continue
                consec = 0
                batch.append((*res, source, post_id))
                preview = (text or "").replace("\n", " ")[:50]
                logger.info("[QW-%s %d/%d] [%s] Score: %+.3f | \"%s...\"", category.upper(), idx, total, source, res[3], preview)

                if len(batch) >= BATCH:

                    batch = []
            if batch:

    async def _score(self, http: httpx.AsyncClient) -> None:
        self._stage("score")
        # FinBERT и Qwen пишутся независимо: сбой одного не отнимает результат другого
        for name, coro in (("finbert", self._score_finbert()), ("qwen", self._score_qwen(http))):
            try:
                await coro
            except Exception as e:
                self._err(f"score {name}", e)

    # ------------------------------------------------------------------ цикл

    async def run_cycle(self, force: bool = False) -> bool:
        """Один полный проход. Если предыдущий ещё идёт — ничего не делает (возвращает False)."""
        if self.lock.locked():
            return False
        async with self.lock:
            t0 = datetime.now(timezone.utc)
            self.status.update(running=True, errors=[])
            try:
                if config.GOLD_DB_PATH and not self._legacy_imported:
                    n = await asyncio.to_thread(posts_db.import_legacy_live, config.GOLD_DB_PATH)
                    self._legacy_imported = True
                    if n:
                        logger.info("Импортировано постов из старой telegram_posts_live: %s", n)

                async with httpx.AsyncClient(timeout=httpx.Timeout(60.0, connect=5.0)) as http:
                    try:
                        await self._fetch_tg()
                    except Exception as e:
                        self._err("fetch telegram", e)
                    try:
                        await self._fetch_rss(http, force)
                    except Exception as e:
                        self._err("fetch rss", e)

                    self._stage("routing")
                    try:
                        # первый цикл после запуска: весь период (подхватывает изменения app/sources.py); дальше — только свежие посты
                        win = None if not self._routed_full else 2
                        await asyncio.to_thread(posts_db.ensure_memberships, win)
                        self._routed_full = True
                        await asyncio.to_thread(posts_db.mark_duplicates)
                    except Exception as e:
                        self._err("routing", e)

                    try:
                        await self._judge(http)
                    except Exception as e:
                        self._err("judge", e)
                    await self._score(http)

                self._stage("aggregate")
                try:
                    await asyncio.to_thread(service.persist_recent)
                except Exception as e:
                    self._err("aggregate", e)
            except Exception as e:
                self._err("cycle", e)
            finally:
                now = datetime.now(timezone.utc)
                self.status.update(
                    running=False, stage="idle",
                    last_cycle_at=now.isoformat(),
                    last_cycle_seconds=round((now - t0).total_seconds(), 1),
                )
            return True

    async def worker(self) -> None:
        """Фоновый воркер: цикл -> пауза -> цикл. Останавливается через task.cancel()."""
        try:
            while True:
                try:
                    await self.run_cycle()
                except asyncio.CancelledError:
                    raise
                except Exception as e:
                    self._err("worker", e)
                await asyncio.sleep(config.TG_POLL_SECONDS)
        finally:
            await self.close()

    async def close(self) -> None:
        if self._client is not None:
            try:
                await self._client.disconnect()
            except Exception:
                pass
            self._client = None
```

```py
# ---------------------------------------------------------------------------
```

```py
# ---------------------------------------------------------------------------

def finbert_scores(text: str) -> tuple[float, float, float, float]:
    """(p_pos, p_neg, p_neu, score). Синхронная — вызывать через asyncio.to_thread."""
    p_pos, p_neg, p_neu = get_finbert().probabilities(text)
    return p_pos, p_neg, p_neu, p_pos - p_neg


# ---------------------------------------------------------------------------
# Qwen: классификатор U / D / N
# ---------------------------------------------------------------------------

def parse_udn(top_logprobs: list[dict]) -> tuple[float, float, float, float] | None:
    """
    top_logprobs первого токена -> (p_bull, p_bear, p_neu, score) либо None, если модель ответила не U/D/N.
    Один и тот же ответ может прийти разными токенами ("U", " U", "u") — их вероятности складываются.
    """
    mass = {"U": 0.0, "D": 0.0, "N": 0.0}
    for item in top_logprobs:
        key = str(item.get("token", "")).strip().upper()
        if key in mass:
            mass[key] += math.exp(item["logprob"])

    total = sum(mass.values())
    if total < 0.5:          # модель ответила не буквой U/D/N — такому результату верить нельзя
        return None
    p_bull, p_bear, p_neu = mass["U"] / total, mass["D"] / total, mass["N"] / total
    return round(p_bull, 4), round(p_bear, 4), round(p_neu, 4), round(p_bull - p_bear, 4)


async def qwen_udn(http: httpx.AsyncClient, text: str, category: str = "general") -> tuple[float, float, float, float] | None:
    """Возвращает (p_bull, p_bear, p_neu, score) или None при ошибке Ollama / нечитаемом ответе."""
    payload = {
        "model": config.QWEN_MODEL,
        "messages": [
            {"role": "system", "content": CATEGORIES[category]["prompt"]},
            {"role": "user", "content": f"Текст новости: {text}\nОценка (U/D/N):"},
        ],
        "max_tokens": 1,
        "temperature": 0.0,
        "logprobs": True,
        "top_logprobs": 10,
    }
    try:
        r = await http.post(f"{config.OLLAMA_URL}/v1/chat/completions", json=payload)
        r.raise_for_status()

        return parse_udn(top)
    except Exception as e:

        await asyncio.sleep(config.QWEN_ERROR_PAUSE)
        return None
```

```py
def _read_history() -> dict:
    """Исторические скоры из news_posts (старый проект, только золото). Кэш на час."""
    now = time.time()

    return vals


def hist_stats(col: str, live_scores: np.ndarray, category: str = "general") -> tuple[float, float, str]:
    """
    (μ, σ, источник). Историю из news_posts используем только для «общего сентимента» (на ней считались скоры по
    золоту); для остальных категорий истории нет — μ = 0, σ по свежим постам этой же категории.
    """
    if category == "general":
        vals = _read_history()[col]
        if len(vals) >= config.MIN_HIST:
            arr = np.asarray(vals, dtype=float)
            return float(arr.mean()), float(arr.std(ddof=1)), f"{config.HIST_TABLE}, n={len(arr)}"
    live = live_scores[~np.isnan(live_scores)]
    if len(live) >= 2 and live.std(ddof=1) > 0:
        return 0.0, float(live.std(ddof=1)), f"свежие посты (истории мало), n={len(live)}"

def _add_combined(table: list[dict], category: str) -> list[dict]:
    """К каждой строке таблицы полупериодов добавляет итоговый скор 'combined' (взвешенно FinBERT + Qwen)."""

            "z": w * f["z"] + (1 - w) * q["z"],

    return table

def build_snapshot(category: str, asof: datetime | None = None) -> dict:
    asof = (asof or datetime.now(timezone.utc)).replace(microsecond=0)

        return base
```

```py
"""
Очистка текста поста и ключ точного дубля.

clean_post  — копия очистки из step1_split_posts.py (убирает «[канал]:», ссылки, хэштеги, подпись
              @Forex_LiveStream). Именно на ТАКОМ тексте считались исторические скоры (step4) и
              обучалась модель, поэтому живые посты оцениваем так же.
norm_key    — копия norm() из step2.py (точный дубль внутри суток).
"""
import re
from collections import OrderedDict

MD_LINK = re.compile(r"\[([^\]]*)\]\(https?://[^)\s]*\)")            # [текст](ссылка) -> текст
ANY_URL = re.compile(r"https?://\S+")
TRAILING_TOKEN = re.compile(r"\s+(#\w+|@Forex_LiveStream|https?://\S+)\s*$", re.IGNORECASE)
SOURCE_PREFIX = re.compile(r"^\[[^\]]+\]:\s*")
SPACES = re.compile(r"\s+")

TRAIL_PUNCT = re.compile(r"[\s.!?…]+$")
QUOTES = str.maketrans({"’": "'", "‘": "'", "“": '"', "”": '"', "«": '"', "»": '"'})


def clean_post(raw: str) -> tuple[str, str | None]:
    """Возвращает (очищенный текст, хэштеги через пробел или None)."""
    body = SOURCE_PREFIX.sub("", raw, count=1)
    body = MD_LINK.sub(r"\1", body)

    tags: list[str] = []
    body = " " + body.rstrip()
    while True:
        m = TRAILING_TOKEN.search(body)
        if not m:
            break
        token = m.group(1)
        if token.startswith("#"):
            tags.append(token)
        body = body[: m.start()]

    body = ANY_URL.sub("", body)
    text = SPACES.sub(" ", body).strip()

    seen: "OrderedDict[str, None]" = OrderedDict()
    for t in reversed(tags):
        seen.setdefault(t, None)
    return text, (" ".join(seen) or None)


def clean_telegram_message(channel: str, text: str) -> tuple[str, str | None]:
    """Так же, как step1b: старый загрузчик хранил пост как '[канал]: текст в одну строку'."""
    return clean_post(f"[{channel}]: " + text.replace("\n", " "))


def norm_key(text: str) -> str:
    t = text.casefold().translate(QUOTES)
    t = SPACES.sub(" ", t).strip()
    return TRAIL_PUNCT.sub("", t)


CYRILLIC = re.compile(r"[\u0400-\u04FF]")
LETTER = re.compile(r"[^\W\d_]")


def cyrillic_share(text: str) -> float:
    """Доля кириллицы среди всех букв текста (0..1). По ней решаем, применим ли англоязычный FinBERT."""
    letters = len(LETTER.findall(text or ""))
    return len(CYRILLIC.findall(text or "")) / letters if letters else 0.0
```

```py
            if pub_time.tzinfo is None:

        except Exception:
            pub_time = now
```

```py
def insert_posts(rows: list[dict]) -> int:
    """
    rows: source, post_id, kind, posted_at, title, text, clean_text, hashtags, url.
    Уже сохранённые пропускаются. Возвращает число новых.
    """
    if not rows:
        return 0
    prepared = []
    for r in rows:
        clean = r.get("clean_text") or ""
        prepared.append((
            r["source"], str(r["post_id"]), r["kind"], r["posted_at"], r.get("title"), r["text"], clean,
            r.get("hashtags"), r.get("url"), norm_key(clean) if clean else None,
            None if clean else 0,          # после очистки ничего не осталось — это не новость
            None if clean else 0,          # (то же для фильтра акций)
        ))
    conn = get_connection()
    before = conn.total_changes
    with conn:
        conn.executemany(
            "INSERT OR IGNORE INTO posts (source, post_id, kind, posted_at, title, text, clean_text, hashtags, url, "
            "norm_key, is_important, is_important_stocks) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)", prepared)
    added = conn.total_changes - before
    conn.close()
    return added

def import_legacy_live(src_path) -> int:
    """
    Разовый импорт из gold_market_normalized.db (таблица telegram_posts_live): текст, время и вердикт Qwen-фильтра.
    Скоры не переносятся (будут посчитаны заново). Нужен только тем, у кого tg_posts ещё не было.

    import sqlite3
    from pathlib import Path
    from app.scoring.text_clean import clean_telegram_message
    p = Path(src_path)
    if not p.exists():
        return 0
    src = sqlite3.connect(f"file:{p.as_posix()}?mode=ro", uri=True, timeout=30)
    try:
        if not src.execute("SELECT 1 FROM sqlite_master WHERE type='table' AND name=?",

    added = insert_posts(rows)
    conn = get_connection()
    with conn:
        conn.executemany("UPDATE posts SET is_important = ? WHERE source = ? AND post_id = ? AND is_important IS NULL",
                         [(v, s, i) for (s, i), v in verdicts.items()])
    conn.close()
    return added

def ensure_memberships(window_days: int | None = None) -> int:
    """
    Для постов за окно определяет, в какие категории они попадают по ТЕКУЩЕЙ конфигурации источников
    (app/sources.py: categories + gated + ключевые слова), и добавляет недостающие строки post_cat.
    Идемпотентно; заодно «доначисляет» категории старым постам, если вы добавили источнику новую категорию.
    """
    window_days = window_days or config.AGG_LOOKBACK_DAYS
    cutoff = _ago(days=window_days)
    src_map = sources.by_key()
    conn = get_connection()
    posts = conn.execute("SELECT source, post_id, clean_text, text FROM posts WHERE posted_at >= ?", (cutoff,)).fetchall()
    have = {(r[0], r[1], r[2]) for r in conn.execute(
        "SELECT c.source, c.post_id, c.category FROM post_cat c JOIN posts p "
        "ON p.source = c.source AND p.post_id = c.post_id WHERE p.posted_at >= ?", (cutoff,))}
    new = []
    for r in posts:
        cfg = src_map.get(r["source"])
        if cfg is None:
            continue
        for cat in sources.categories_for(cfg, r["clean_text"] or r["text"]):
            if (r["source"], r["post_id"], cat) not in have:
                new.append((r["source"], r["post_id"], cat))
    if new:
        with conn:
            conn.executemany("INSERT OR IGNORE INTO post_cat (source, post_id, category) VALUES (?,?,?)", new)
    conn.close()
    return len(new)


def mark_duplicates(window_days: int | None = None) -> int:
    """
    Точные дубли внутри суток (правило step2): остаётся самый ранний пост, остальные is_dup = 1. Идемпотентно.

    Акции: одну и ту же новость отдают сразу Yahoo, Google News, CNBC, MarketWatch … и ленты разных тикеров.
    Чтобы дубль не «украл» новость у тикера, чей источник пришёл позже, категории АКЦИЙ дубля переносятся
    на оставшийся (самый ранний) пост — он оценивается Qwen и для них. Категории золота не трогаем.
    """
    cutoff = _ago(days=window_days or config.DEDUP_WINDOW_DAYS)
    conn = get_connection()
    rows = conn.execute(
        "SELECT source, post_id, posted_at, norm_key, is_dup FROM posts "
        "WHERE posted_at >= ? AND norm_key IS NOT NULL AND norm_key <> ''", (cutoff,)).fetchall()
    groups: dict[tuple, list] = defaultdict(list)
    for r in rows:
        groups[(r["posted_at"][:10], r["norm_key"])].append(r)
    updates, merges = [], []
    stock_cats = set(STOCK_ORDER)
    dup_cats: dict[tuple, list] = defaultdict(list)       # (source, post_id) дубля -> его категории акций
    if stock_cats:
        for c in conn.execute("SELECT c.source, c.post_id, c.category FROM post_cat c JOIN posts p "
                              "ON p.source = c.source AND p.post_id = c.post_id WHERE p.posted_at >= ?", (cutoff,)):
            if c["category"] in stock_cats:
                dup_cats[(c["source"], c["post_id"])].append(c["category"])
    for g in groups.values():
        g.sort(key=lambda r: (r["posted_at"], r["source"], r["post_id"]))
        for i, r in enumerate(g):
            want = 0 if i == 0 else 1
            if r["is_dup"] != want:
                updates.append((want, r["source"], r["post_id"]))
            if i > 0:
                for cat in dup_cats.get((r["source"], r["post_id"]), ()):
                    merges.append((g[0]["source"], g[0]["post_id"], cat))
    if updates or merges:
        with conn:
            if updates:
                conn.executemany("UPDATE posts SET is_dup = ? WHERE source = ? AND post_id = ?", updates)
            if merges:
                conn.executemany("INSERT OR IGNORE INTO post_cat (source, post_id, category) VALUES (?,?,?)", merges)
    conn.close()
    return len(updates)

def recent_posts(category: str, limit: int = 40, hours: int | None = None) -> list[dict]:
    sql = ("SELECT p.source, p.kind, p.posted_at, p.title, p.text, p.clean_text, p.url, "
           "p.finbert_pos, p.finbert_neg, p.finbert_neu, p.finbert_score, "
           "c.qwen_bull, c.qwen_bear, c.qwen_neu, c.qwen_score FROM post_cat c "
           "JOIN posts p ON p.source = c.source AND p.post_id = c.post_id "
           f"WHERE c.category = ? AND {_imp(category)} = 1 AND p.is_dup = 0 ")
    args: list = [category]
    if hours:
        sql += "AND p.posted_at >= ? "
        args.append(_ago(hours=hours))
    sql += "ORDER BY p.posted_at DESC LIMIT ?"
    args.append(limit)
    conn = get_connection()
    rows = conn.execute(sql, args).fetchall()
    conn.close()
    out = []
    for r in rows:
        d = dict(r)
        d["display_text"] = d.pop("clean_text") or d["text"]
        d.pop("text", None)
        out.append(d)
    return out


def counts_global() -> dict:
    """Счётчики для всего конвейера. «Очередь фильтра» = золотая + акционная очередь; «мусор» — отклонённые всеми фильтрами."""
    conn = get_connection()
    r = conn.execute(
        f"""SELECT COUNT(*) AS total,
                   COALESCE(SUM(is_dup = 1), 0) AS dups,
                   COALESCE(SUM(is_dup = 0 AND (is_important IS NOT NULL OR is_important_stocks IS NOT NULL)
                                AND COALESCE(is_important, 0) = 0 AND COALESCE(is_important_stocks, 0) = 0), 0) AS trash,
                   COALESCE(SUM(is_dup = 0 AND ((is_important IS NULL AND {_HAS_GOLD.replace('p.', 'posts.')})
                                OR (is_important_stocks IS NULL AND {_HAS_STOCK.replace('p.', 'posts.')}))), 0) AS pending_judge,
                   COALESCE(SUM((is_important = 1 OR is_important_stocks = 1) AND is_dup = 0
                                AND finbert_score IS NULL AND finbert_neu IS NULL), 0) AS pending_finbert,
                   MIN(posted_at) AS first_post, MAX(posted_at) AS last_post
            FROM posts""").fetchone()
    pq = conn.execute(
        "SELECT COUNT(*) FROM post_cat c JOIN posts p ON p.source = c.source AND p.post_id = c.post_id "
        f"WHERE c.qwen_score IS NULL AND p.is_dup = 0 AND ((c.category IN ({_GOLD_CATS}) AND p.is_important = 1) "
        f"OR (c.category IN ({_STOCK_CATS}) AND p.is_important_stocks = 1))").fetchone()[0]
    conn.close()
    d = dict(r)
    d["pending_qwen"] = pq
    return d


def counts_category(category: str) -> dict:
    imp = _imp(category)
    conn = get_connection()
    r = conn.execute(
        f"""SELECT COUNT(*) AS members,
                  COALESCE(SUM({imp} = 1 AND p.is_dup = 0), 0) AS important,
                  COALESCE(SUM({imp} = 0 AND p.is_dup = 0), 0) AS trash,
                  COALESCE(SUM(p.is_dup = 1), 0) AS dups,
                  COALESCE(SUM({imp} IS NULL AND p.is_dup = 0), 0) AS pending_judge,
                  COALESCE(SUM({imp} = 1 AND p.is_dup = 0 AND {_FB_PENDING}), 0) AS pending_finbert,
                  COALESCE(SUM({imp} = 1 AND p.is_dup = 0 AND c.qwen_score IS NULL), 0) AS pending_qwen,
                  COALESCE(SUM({imp} = 1 AND p.is_dup = 0 AND p.posted_at >= ?), 0) AS posts_24h,
                  MIN(CASE WHEN {imp} = 1 AND p.is_dup = 0 THEN p.posted_at END) AS first_post,
                  MAX(CASE WHEN {imp} = 1 AND p.is_dup = 0 THEN p.posted_at END) AS last_post
           FROM post_cat c JOIN posts p ON p.source = c.source AND p.post_id = c.post_id
           WHERE c.category = ?""", (_ago(hours=24), category)).fetchone()
    conn.close()
    return dict(r)


def source_stats(category: str, days: int = 7) -> dict:
    """source -> {'posts': N, 'last_post': ts} среди важных постов категории за последние `days` суток."""
    conn = get_connection()
    rows = conn.execute(
        "SELECT p.source, COUNT(*) AS n, MAX(p.posted_at) AS last FROM post_cat c "
        "JOIN posts p ON p.source = c.source AND p.post_id = c.post_id "
        f"WHERE c.category = ? AND {_imp(category)} = 1 AND p.is_dup = 0 AND p.posted_at >= ? GROUP BY p.source",
        (category, _ago(days=days))).fetchall()
    conn.close()
    return {r["source"]: {"posts": r["n"], "last_post": r["last"]} for r in rows}

def save_asof_rows(category: str, rows: list[dict]) -> None:
    if not rows:
        return
    conn = get_connection()
    conn.execute(f"CREATE TABLE IF NOT EXISTS {ASOF_TABLE} (category TEXT NOT NULL, asof TEXT NOT NULL, "
                 "PRIMARY KEY (category, asof))")
    existing = {r[1] for r in conn.execute(f"PRAGMA table_info({ASOF_TABLE})")}
    for col in rows[0]:
        if col != "asof" and col not in existing:
            conn.execute(f"ALTER TABLE {ASOF_TABLE} ADD COLUMN {col} {'INTEGER' if col.endswith('_n') else 'REAL'}")
    names = ["category"] + list(rows[0].keys())
    sql = f"INSERT OR REPLACE INTO {ASOF_TABLE} ({', '.join(names)}) VALUES ({', '.join('?' * len(names))})"
    with conn:
        conn.executemany(sql, [(category, *[r[n] for n in names[1:]]) for r in rows])
    conn.close()
```

```html
            else warn.classList.add('hidden');

                    `<div class="rounded-xl p-4 bg-black/30 border border-cardborder md:col-span-2 flex items-center text-sm text-gray-500">

        function renderChart(s) {

        function renderPosts(posts) {
            const box = document.getElementById('bg-posts');

'title="FinBERT понимает только английский — для этого поста пропущен"' : ''}><span class="text-gray-500">${label}</span><span class="${v === null || v === undefined ? 'text-gray-600' : color} font-bold mt-0.5">${v === null || v === undefined ? (skipped ? '—' : '…') : v.toFixed(2)}</span></span>`;

                            <span class="text-[11px] text-gray-400 font-mono">${parseUtc(p.posted_at).toLocaleString('ru-RU', {day: '2-digit', month: '2-digit', hour: '2-digit', minute: '2-digit'})}</span>

                    </div></div>`;
            }).join('');

                        <span class="font-semibold ${s.kind === 'tg' ? 'text-cyan-400' : 'text-indigo-300'} truncate">${url ? `<a href="${escapeHtml(url)}" target="_blank" rel="noopener noreferrer" class="hover:underline">${escapeHtml(name)}</a>` : escapeHtml(name)}</span>

                    </div>

        }

        // ---------- загрузка ----------
        async function load() {
            try {
                const base = '/api/v1/category/' + encodeURIComponent(CAT_KEY);
                const [snapRes, serRes, postsRes] = await Promise.all([fetch(base), fetch(base + '/series?hours=' + SERIES_HOURS), fetch(base + '/posts?limit=40')]);
                if (snapRes.status === 404) { document.getElementById('cat-title').textContent = 'Категория не найдена'; return; }
                snapshot = await snapRes.json(); series = await serRes.json();
                const posts = await postsRes.json();
                renderStatus(snapshot.pipeline || {});
                renderSnapshot(snapshot); renderChart(series); renderPosts(posts);
                renderSources(snapshot.sources, (snapshot.pipeline || {}).sources);
                clearTimeout(pollTimer);
                if (snapshot.pipeline && snapshot.pipeline.running) pollTimer = setTimeout(load, 5000);
            } catch (err) {
                console.error('Ошибка загрузки категории:', err);
                const el = document.getElementById('bg-status');
                el.className = 'px-2.5 py-1 text-[11px] font-semibold rounded border bg-rose-500/10 text-rose-400 border-rose-500/20';
                el.textContent = 'Ошибка загрузки API';
            }
        }

        async function refreshPipeline() {
            const btn = document.getElementById('bg-refresh-btn');
            btn.disabled = true; btn.classList.add('opacity-60');
            try { await fetch('/api/v1/pipeline/refresh', { method: 'POST' }); setTimeout(load, 1500); }
            finally { setTimeout(() => { btn.disabled = false; btn.classList.remove('opacity-60'); }, 3000); }
        }

        lucide.createIcons();
        load();
        setInterval(load, 60000);
    </script>
</body>
</html>
```

```py
"""
Проверка источников из app/sources.py и подбор Telegram-каналов по РЕАЛЬНОМУ содержимому, а не по названию.

    python scripts/check_sources.py                       # RSS-ленты: открылись ли, сколько свежих записей, куда маршрутизируются
    python scripts/check_sources.py --only stocks         # только ленты акций (или --only gold)
    python scripts/check_sources.py --tg                  # каналы из sources.py: активность, доля «по тикерам», язык, реклама
    python scripts/check_sources.py --probe marketfeed https://t.me/markettwits    # то же для любых каналов (ещё не добавленных)
    python scripts/check_sources.py --find-tg             # поиск каналов по стандартным запросам + оценка содержимого
    python scripts/check_sources.py --find-tg "nasdaq" "earnings" --min-subs 1000 --top 20

Что считается по каналу (последние --limit постов, по умолчанию 100):
  постов/день     как часто пишут
  по тикерам %    какая доля постов попала бы в акции из app/stocks.py (ключевые слова / тикеры — как в боевом конвейере)
  тикеров/день    постов/день × доля — то, что реально дойдёт до Qwen. Канал с 300 постами в день и 0.5 по теме
                  не нужен; канал с 5 постами в день и 60% по теме — нужен
  рус %           доля русскоязычных постов (для них FinBERT пропускается, итог только по Qwen)
  реклама %       «VIP», «сигналы», «подписывайтесь», «промокод» …

ВАЖНО для --tg / --probe / --find-tg: одну сессию Telegram не могут держать две программы. Остановите сайт (uvicorn)
или укажите копию сессии: TG_SESSION=<путь к копии без .session>. Между каналами скрипт делает паузы (защита от FloodWait).
"""

STOCK_CATS = set(TICKERS) | {MARKET_KEY}

DEFAULT_QUERIES = ["US stock market", "wall street news", "stock market news", "earnings news", "nasdaq", "S&P 500",
                   "акции США", "фондовый рынок США"]
PROMO = re.compile(
    r"\bvip\b|\bsignals?\b|premium|join (us|now|our)|subscribe|promo ?code|giveaway|referral|bonus|\bcopy ?trad|"
    r"free (course|webinar)|сигнал|подпис|промокод|бонус|розыгрыш|курс по|обучени|вип[- ]?(канал|группа)|t\.me/\+",
    re.IGNORECASE)
GOOD_PER_DAY = 1.0          # меньше «тикерных» постов в сутки — канал почти бесполезен для сентимента
MAX_PROMO = 0.25


def _age(dt: datetime | None) -> str:
    if dt is None:
        return "—"
    h = (datetime.now(timezone.utc) - dt).total_seconds() / 3600
    return f"{h * 60:.0f} мин" if h < 1.5 else f"{h:.1f} ч" if h < 48 else f"{h / 24:.0f} дн"


def norm_channel(name: str) -> str:

    name = name.strip()

# ---------------------------------------------------------------------------
# Оценка канала по его постам (чистая функция — проверяется в scripts/selftest.py без сети)

def analyse(messages: list[tuple[datetime, str]], channel: str, cfg: dict | None = None) -> dict:
    """messages: (время UTC, сырой текст). cfg — конфиг источника (по умолчанию: «широкий» канал ALL + US с фильтром по словам)."""
    cfg = cfg or {"id": channel, "kind": "tg", "categories": ALL + US, "gated": ALL + US}
    n = len(messages)
    out = {"n": n, "per_day": 0.0, "stock_hit": 0.0, "gold_hit": 0.0, "stock_per_day": 0.0, "ru": 0.0, "promo": 0.0,
           "top": [], "newest": None, "verdict": "нет постов"}
    if not n:
        return out
    stock = gold = ru = promo = 0
    tickers: dict[str, int] = {}
    for _, text in messages:
        clean, _tags = clean_telegram_message(channel, text)
        cats = sources.categories_for(cfg, clean)
        hit = [c for c in cats if c in STOCK_CATS]
        stock += bool(hit)
        gold += any(c in ALL for c in cats)
        for c in hit:
            tickers[c] = tickers.get(c, 0) + 1
        ru += cyrillic_share(clean) >= config.CYRILLIC_THRESHOLD
        promo += bool(PROMO.search(text))
    dates = [d for d, _ in messages]
    span_days = max((max(dates) - min(dates)).total_seconds() / 86400, 1 / 24)
    out.update(per_day=n / span_days, stock_hit=stock / n, gold_hit=gold / n, ru=ru / n, promo=promo / n,
               newest=max(dates), top=[f"{k}:{v}" for k, v in sorted(tickers.items(), key=lambda kv: -kv[1])[:4]])
    out["stock_per_day"] = out["per_day"] * out["stock_hit"]
    if out["promo"] > MAX_PROMO:
        out["verdict"] = "🚫 реклама / сигналы"
    elif out["stock_per_day"] < GOOD_PER_DAY:
        out["verdict"] = "⚠️ мало по теме"
    else:
        out["verdict"] = "✅ годится"
    return out


def stats_line(name: str, subs, a: dict) -> str:
    subs_s = "?" if subs is None else str(subs)
    return (f"@{name:<26} подп. {subs_s:>8} | постов/д {a['per_day']:>6.1f} | по тикерам {a['stock_hit'] * 100:>3.0f}% "
            f"({a['stock_per_day']:>5.1f}/д) | золото {a['gold_hit'] * 100:>3.0f}% | рус {a['ru'] * 100:>3.0f}% | "
            f"реклама {a['promo'] * 100:>3.0f}% | {a['verdict']}"
            + (f" | {', '.join(a['top'])}" if a["top"] else ""))

def check_rss(only: str | None) -> int:
    import feedparser
    import httpx
    from app.collector.rss_ingest import HEADERS, parse_feed_text

    feeds = [s for s in sources.rss_sources()
             if only is None or (only == "stocks") == bool(set(s["categories"]) & STOCK_CATS)]
    print(f"RSS-лент к проверке: {len(feeds)}\n")

    def one(src: dict) -> dict:
        try:
            with httpx.Client(headers=HEADERS, follow_redirects=True, timeout=config.RSS_TIMEOUT) as c:
                r = c.get(src["url"])
            r.raise_for_status()
            raw = feedparser.parse(r.text)
            rows = parse_feed_text(src, r.text)
            routed: dict[str, int] = {}
            for row in rows:
                for cat in sources.categories_for(src, row["clean_text"] or row["text"]):
                    routed[cat] = routed.get(cat, 0) + 1
            newest = max((datetime.strptime(x["posted_at"], "%Y-%m-%d %H:%M:%S").replace(tzinfo=timezone.utc)
                          for x in rows), default=None)
            return {"src": src, "ok": True, "raw": len(raw.entries), "fresh": len(rows), "routed": routed, "newest": newest}
        except Exception as e:
            return {"src": src, "ok": False, "error": f"{type(e).__name__}: {e}"[:160]}

    with ThreadPoolExecutor(max_workers=6) as ex:
        results = list(ex.map(one, feeds))

    good = 0
    for r in results:
        sid = r["src"]["id"]
        if not r["ok"]:
            print(f"❌ {sid:<26} {r['error']}\n   {r['src']['url']}")
            continue
        warn = r["raw"] == 0 or r["fresh"] == 0
        good += not warn
        routed = ", ".join(f"{k}:{v}" for k, v in sorted(r["routed"].items())) or "никуда"
        print(f"{'⚠️ ' if warn else '✅'} {sid:<26} записей {r['raw']:>3}, свежих {r['fresh']:>3}, последняя {_age(r['newest']):>7} | попадёт: {routed}")
        if warn:
            print(f"   лента пуста или всё старше {config.RSS_MAX_AGE_HOURS} ч: {r['src']['url']}")
    print(f"\nИтого рабочих: {good} из {len(feeds)}")
    return len(feeds) - good

async def _messages(client, entity, limit: int) -> list[tuple[datetime, str]]:
    out = []
    async for m in client.iter_messages(entity, limit=limit):
        if m.text:
            out.append((m.date.astimezone(timezone.utc), m.text))
    return out


async def check_tg(mode: str, names: list[str], min_subs: int, top: int, limit: int) -> None:
    if not (config.TG_API_ID and config.TG_API_HASH):
        sys.exit("Задайте TG_API_ID и TG_API_HASH (и сделайте вход: python scripts/tg_login.py).")
    from telethon import TelegramClient
    from telethon.errors import FloodWaitError
    from telethon.tl.functions.contacts import SearchRequest

    client = TelegramClient(config.TG_SESSION, int(config.TG_API_ID), config.TG_API_HASH)
    await client.connect()
    try:
        if not await client.is_user_authorized():
            sys.exit("Нет входа в Telegram: python scripts/tg_login.py")

        async def probe(name: str, cfg: dict | None, subs=None, entity=None):
            ent = entity or await client.get_entity(name)
            if subs is None:
                subs = getattr(ent, "participants_count", None)
            msgs = await _messages(client, ent, limit)
            return subs, analyse(msgs, name, cfg)

        if mode == "sources":                                    # каналы из app/sources.py — по их боевой конфигурации
            print(f"Каналов в sources.py: {len(sources.tg_sources())} (оценка по последним {limit} постам)\n")
            for s in sources.tg_sources():
                try:
                    subs, a = await probe(s["id"], s)
                    print(stats_line(s["id"], subs, a) + f" | последний пост {_age(a['newest'])} назад")
                except FloodWaitError as e:
                    sys.exit(f"Telegram просит подождать {e.seconds} с (FloodWait) — повторите позже.")
                except Exception as e:
                    print(f"❌ @{s['id']:<26} {type(e).__name__}: {e}"[:200])
                await asyncio.sleep(1.0)

        elif mode == "probe":                                    # произвольные каналы — как «широкий» источник ALL + US
            for raw in names:
                name = norm_channel(raw)
                try:
                    subs, a = await probe(name, None)
                    print(stats_line(name, subs, a) + f" | последний пост {_age(a['newest'])} назад")
                except FloodWaitError as e:
                    sys.exit(f"Telegram просит подождать {e.seconds} с (FloodWait) — повторите позже.")
                except Exception as e:
                    print(f"❌ @{name:<26} {type(e).__name__}: {e}"[:200])
                await asyncio.sleep(1.0)

        else:                                                    # поиск
            queries = names or DEFAULT_QUERIES
            have = {s["id"].lower() for s in sources.tg_sources()}
            found: dict[str, object] = {}
            for q in queries:
                res = await client(SearchRequest(q=q, limit=30))
                for ch in res.chats:
                    u = getattr(ch, "username", None)
                    if not (getattr(ch, "broadcast", False) and u) or u.lower() in have:
                        continue
                    subs = getattr(ch, "participants_count", None)
                    if subs is not None and subs < min_subs:
                        continue
                    found[u] = ch
                await asyncio.sleep(1.0)
            cands = sorted(found.values(), key=lambda c: -(getattr(c, "participants_count", 0) or 0))[:top]
            print(f"Найдено каналов (≥{min_subs} подписчиков, ещё не в sources.py): {len(found)}; оцениваю {len(cands)} крупнейших "
                  f"по последним {limit} постам…\n")
            scored = []
            for ch in cands:
                try:
                    subs, a = await probe(ch.username, None, getattr(ch, "participants_count", None), ch)
                    scored.append((a["stock_per_day"] if a["verdict"].startswith("✅") else -1, ch.username, subs, a))
                except FloodWaitError as e:
                    print(f"Telegram просит подождать {e.seconds} с (FloodWait) — показываю то, что успел.")
                    break
                except Exception as e:
                    print(f"❌ @{ch.username}: {type(e).__name__}: {e}"[:160])
                await asyncio.sleep(1.5)
            for _, name, subs, a in sorted(scored, key=lambda x: -x[0]):
                print(stats_line(name, subs, a))
            print('\nПодходящий канал (✅) добавьте в app/sources.py, например:\n'
                  '   {"id": "<username>", "kind": "tg", "categories": US, "gated": US},')
    finally:
        await client.disconnect()


def main() -> None:
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument("--only", choices=["stocks", "gold"], help="RSS: проверять только ленты акций / золота")
    ap.add_argument("--tg", action="store_true", help="оценить Telegram-каналы из sources.py")
```

```py
"""
Проверка скоров панели на реальных ценах: «предсказывает ли скор движение рынка?»

Что делает
----------
1. Берёт посты из базы (posts / post_cat) и ПЕРЕСЧИТЫВАЕТ скоры на каждый целый час в прошлом тем же кодом,
   что и панель (app/scoring/aggregate.py). В расчёт идут только посты с posted_at <= момента расчёта.
2. Берёт почасовые цены: yfinance (золото GC=F, нефть CL=F, US10Y ^TNX, доллар DX-Y.NYB) или ваш CSV.
   Цена «известна» только после закрытия часового бара — заглядывания в будущее нет.
3. Для каждого часа считает, как изменилась цена через H часов, и сравнивает со скором:
     IC      ранговая корреляция (Спирмен) «скор -> будущее движение». > 0: скор угадывает направление.
     95% CI  доверительный интервал IC (блочный бутстрэп: учитывает, что окна H часов перекрываются).
     hit%    доля верных знаков среди «не нейтральных» сигналов (|скор| >= порога).
     спред   средний ход цены после бычьего сигнала минус после медвежьего, в б.п.
4. Сравнивает: FinBERT, Qwen, арифметическое среднее и знаковое взвешенное геометрическое среднее.
   Дополнительно: IC по всем полупериодам (3/12/24/72 ч) и (с --sweep) по сетке весов FinBERT/Qwen.

Знак скора = направление ЦЕНЫ актива (для US10Y — направление ДОХОДНОСТИ), поэтому IC > 0 всегда «хорошо».

Запуск (из корня проекта, панель может работать параллельно):

    python scripts/validate_scores.py --csv general=C:/data/XAUUSD_H1.csv        # свои цены вместо yfinance

Формат CSV: первая колонка — время бара (UTC), колонка Close/close/price (или последняя). Если время в файле
брокерское (например UTC+3), укажите --csv-shift-h 3. Время считается ОТКРЫТИЕМ часового бара (--bar-hours 1).

def arith_mean(f, q, w):
    return w * np.asarray(f, dtype=float) + (1.0 - w) * np.asarray(q, dtype=float)


def geo_mean(f, q, w):
    """Знаковое взвешенное геометрическое среднее: знаки совпали -> sign·|f|^w·|q|^(1-w), иначе 0."""
    f = np.asarray(f, dtype=float)
    q = np.asarray(q, dtype=float)
    out = np.zeros_like(f)
    ok = (f * q) > 0
    out[ok] = np.sign(f[ok]) * np.abs(f[ok]) ** w * np.abs(q[ok]) ** (1.0 - w)
    return out


def make_signals(fb, qw, w):
    """kind -> (название, массив)."""
    return {
        "fb": ("FinBERT", np.asarray(fb, dtype=float)),
        "qw": ("Qwen", np.asarray(qw, dtype=float)),
        "arith": (f"Арифм. {w:.0%}/{1 - w:.0%}", arith_mean(fb, qw, w)),
        "geo": (f"Геометр. {w:.0%}/{1 - w:.0%}", geo_mean(fb, qw, w)),

def to_ns(idx) -> np.ndarray:
    """DatetimeIndex (любой tz / единицы времени) -> int64 наносекунды UTC."""
    idx = pd.DatetimeIndex(idx)
    idx = idx.tz_localize("UTC") if idx.tz is None else idx.tz_convert("UTC")
    return idx.tz_localize(None).to_numpy(dtype="datetime64[ns]").astype("int64")


def load_prices(category, start, end, csv_map, bar_hours, csv_shift_h) -> pd.Series:
    """Series: индекс = момент, когда цена стала известна (конец бара), значения = Close."""
    if category in csv_map:
        df = pd.read_csv(csv_map[category])
        tcol = df.columns[0]
        ccol = next((c for c in df.columns if str(c).lower() in ("close", "c", "price")), df.columns[-1])
        idx = pd.DatetimeIndex(pd.to_datetime(df[tcol], utc=True)) - pd.Timedelta(hours=csv_shift_h)
        s = pd.Series(df[ccol].astype(float).to_numpy(), index=idx)
    else:
        import yfinance as yf
        asset = ASSETS[category]
        df = yf.download(asset["ticker"],
                         start=(start - pd.Timedelta(days=2)).strftime("%Y-%m-%d"),
                         end=(end + pd.Timedelta(days=1)).strftime("%Y-%m-%d"),
                         interval="1h", progress=False, auto_adjust=False)
        if df is None or df.empty:
            raise RuntimeError(f"yfinance не вернул часовые данные по {asset['ticker']}")
        close = df["Close"]
        if isinstance(close, pd.DataFrame):          # новые версии yfinance дают MultiIndex-колонки
            close = close.iloc[:, 0]
        idx = pd.DatetimeIndex(close.index)
        idx = idx.tz_localize("UTC") if idx.tz is None else idx.tz_convert("UTC")
        s = pd.Series(close.astype(float).to_numpy(), index=idx)
        bar_hours = 1.0
    s = s.dropna()
    s.index = s.index + pd.Timedelta(hours=bar_hours)        # цена известна в КОНЦЕ бара
    s = s[~s.index.duplicated(keep="last")].sort_index()
    if len(s) < 50:
        raise RuntimeError(f"слишком мало ценовых баров: {len(s)}")
    return s


def price_at(bar_end_ns, prices, times_ns, max_stale_ns):
    """Последняя известная цена на каждый момент; NaN, если свежего бара нет (рынок закрыт / выходные)."""
    pos = np.searchsorted(bar_end_ns, times_ns, side="right") - 1
    have = pos >= 0
    safe = np.clip(pos, 0, None)
    out = np.where(have, prices[safe], np.nan)
    age = times_ns - bar_end_ns[safe]
    out[(~have) | (age > max_stale_ns)] = np.nan
    return out


def forward_returns(scores_asof, prices: pd.Series, horizon_h, kind, max_stale_h):
    """Изменение цены за horizon_h часов после момента расчёта скора, в б.п. (NaN, если рынок был закрыт)."""
    bar_end = to_ns(prices.index)
    px = prices.to_numpy(dtype=float)
    t0 = to_ns(scores_asof)
    t1 = t0 + int(horizon_h * 3600 * 1e9)
    stale = int(max_stale_h * 3600 * 1e9)
    p0, p1 = price_at(bar_end, px, t0, stale), price_at(bar_end, px, t1, stale)
    with np.errstate(divide="ignore", invalid="ignore"):
        if kind == "bps":
            return (p1 - p0) * 100.0
        return np.log(p1 / p0) * 1e4

def spearman(x, y):
    return pd.Series(x).rank().corr(pd.Series(y).rank())


def block_ci(x, y, block, boots, seed=1):
    """95% доверительный интервал IC блочным бутстрэпом (блоки подряд идущих точек)."""
    n = len(x)
    if boots <= 0 or n < 3 * block:
        return np.nan, np.nan
    rng = np.random.default_rng(seed)
    nb = int(np.ceil(n / block))
    vals = []
    for _ in range(boots):
        starts = rng.integers(0, n - block + 1, size=nb)
        idx = (starts[:, None] + np.arange(block)[None, :]).ravel()[:n]
        vals.append(spearman(x[idx], y[idx]))
    vals = np.asarray(vals, dtype=float)
    if np.isnan(vals).all():
        return np.nan, np.nan
    lo, hi = np.nanpercentile(vals, [2.5, 97.5])
    return float(lo), float(hi)


def evaluate(sig, ret, horizon, thr, boots):
    ok = np.isfinite(sig) & np.isfinite(ret)
    sig, ret = sig[ok], ret[ok]
    n = len(sig)
    res = {"n": n, "n_eff": n / max(horizon, 1), "ic": np.nan, "ci_lo": np.nan, "ci_hi": np.nan,
           "hit": np.nan, "trades": 0, "spread": np.nan}
    if n < 20:
        return res
    res["ic"] = float(spearman(sig, ret))
    res["ci_lo"], res["ci_hi"] = block_ci(sig, ret, max(horizon, 24), boots)
    act = (np.abs(sig) >= thr) & (ret != 0)
    res["trades"] = int(act.sum())
    if act.sum() >= 10:
        res["hit"] = float((np.sign(sig[act]) == np.sign(ret[act])).mean())
    bull, bear = ret[sig >= thr], ret[sig <= -thr]
    if len(bull) >= 5 and len(bear) >= 5:
        res["spread"] = float(bull.mean() - bear.mean())
    return res

    p.add_argument("--horizons", default="6,24,72", help="горизонты, часы, через запятую")
    p.add_argument("--hl", type=float, default=float(config.PRIMARY_HL), help="основной полупериод, ч")
    p.add_argument("--w", type=float, default=config.ENSEMBLE_W_FINBERT, help="вес FinBERT (как ENSEMBLE_W_FINBERT)")
    p.add_argument("--threshold", type=float, default=config.SIGNAL_THRESHOLD, help="порог |скор| для hit%% и спреда")
    p.add_argument("--lookback-days", type=int, default=config.AGG_LOOKBACK_DAYS, help="сколько суток постов брать")
    p.add_argument("--warmup-h", type=float, default=48.0, help="сколько часов после первого поста пропустить")
    p.add_argument("--max-stale-h", type=float, default=1.5, help="цена старше этого = рынок закрыт, точка пропускается")
    p.add_argument("--boots", type=int, default=300, help="число бутстрэп-выборок (0 = без доверительных интервалов)")
    p.add_argument("--sweep", action="store_true", help="перебрать веса FinBERT/Qwen от 0 до 1 с шагом 0.1")
    p.add_argument("--csv", action="append", default=[], metavar="CAT=PATH", help="свои цены: general=path.csv")
    p.add_argument("--bar-hours", type=float, default=1.0, help="длина бара в CSV (0 = время уже и есть момент цены)")
    p.add_argument("--csv-shift-h", type=float, default=0.0, help="часов вычесть из времени CSV, чтобы получить UTC")
    p.add_argument("--out", default="validate_scores_results.csv", help="куда сохранить все результаты")
    return p.parse_args()


def main():
    args = parse_args()
    cats = [c.strip() for c in args.categories.split(",") if c.strip()]
    unknown = [c for c in cats if c not in ASSETS]
    if unknown:
        sys.exit(f"Неизвестные категории: {unknown}. Доступны: {list(ASSETS)}")
    horizons = sorted({int(x) for x in args.horizons.split(",") if x.strip()})
    csv_map = dict(item.split("=", 1) for item in args.csv)
    half_lives = sorted(set(config.HALF_LIVES_H) | {args.hl})
    w = args.w

    try:
        posts_db.init_db()
    except Exception as e:                              # база могла быть занята — не критично
        print(f"(init_db: {e})")

    all_rows, sweep_rows = [], []
    for cat in cats:
        asset = ASSETS[cat]
        print("=" * 100)
        print(f"{cat} · {asset['name']} · основной полупериод {args.hl:g} ч")
        print("=" * 100)

        scores = build_scores(cat, args.lookback_days, args.warmup_h, half_lives)
        if scores is None or len(scores) < 24:
            print("Недостаточно скоренных постов (или данных меньше разогрева). Дайте конвейеру поработать "
                  "или уменьшите --warmup-h.\n")
            continue
        try:
            prices = load_prices(cat, scores["asof"].iloc[0], scores["asof"].iloc[-1], csv_map,
                                 args.bar_hours, args.csv_shift_h)
        except Exception as e:
            print(f"Не удалось получить цены: {type(e).__name__}: {e}\n"
                  f"Подсказка: yfinance/сеть, либо передайте свои цены: --csv {cat}=путь.csv\n")
            continue

        rets = {h: forward_returns(scores["asof"], prices, h, asset["kind"], args.max_stale_h) for h in horizons}
        print(f"Скоры: {scores['asof'].iloc[0]:%Y-%m-%d %H:%M} … {scores['asof'].iloc[-1]:%Y-%m-%d %H:%M} UTC, "
              f"{len(scores)} часов. Цены: {len(prices)} баров. "
              "Точек с известной ценой в начале и в конце окна: "
              + ", ".join(f"{h}ч — {int(np.isfinite(rets[h]).sum())}" for h in horizons) + "\n")

        cat_rows = []
        for hl in half_lives:
            fb, qw = scores[f"fb_{hl:g}"].to_numpy(), scores[f"qw_{hl:g}"].to_numpy()
            signals = make_signals(fb, qw, w)
            for h in horizons:
                for kind, (name, arr) in signals.items():
                    boots = args.boots if abs(hl - args.hl) < 1e-9 else 0
                    r = evaluate(arr, rets[h], h, args.threshold, boots)
                    r.update(category=cat, hl=hl, horizon=h, kind=kind, signal=name, w=w)
                    cat_rows.append(r)

        main_rows = [r for r in cat_rows if abs(r["hl"] - args.hl) < 1e-9]
        print_main_table(main_rows, horizons)
        print_hl_pivot(cat_rows, "arith", f"арифметическое {w:.0%}/{1 - w:.0%}", horizons, half_lives)
        print_hl_pivot(cat_rows, "geo", f"геометрическое {w:.0%}/{1 - w:.0%}", horizons, half_lives)
        all_rows += cat_rows

        if args.sweep:
            fb, qw = scores[f"fb_{args.hl:g}"].to_numpy(), scores[f"qw_{args.hl:g}"].to_numpy()
            cat_sweep = []
            for ww in np.round(np.arange(0.0, 1.0001, 0.1), 1):
                sigs = make_signals(fb, qw, float(ww))
                for h in horizons:
                    for kind in ("arith", "geo"):
                        r = evaluate(sigs[kind][1], rets[h], h, args.threshold, 0)
                        r.update(category=cat, hl=args.hl, horizon=h, kind=kind, signal=sigs[kind][0], w=float(ww))
                        cat_sweep.append(r)
            print_sweep(cat_sweep, horizons)
            sweep_rows += cat_sweep

    if all_rows:
        df = pd.DataFrame(all_rows + sweep_rows)
        df["sweep"] = [False] * len(all_rows) + [True] * len(sweep_rows)
        cols = ["category", "hl", "horizon", "kind", "signal", "w", "sweep", "n", "n_eff", "ic", "ci_lo", "ci_hi",
                "hit", "trades", "spread"]
        df[cols].to_csv(args.out, index=False, encoding="utf-8-sig")
        print(f"Полные результаты сохранены: {Path(args.out).resolve()}\n")
        print(LEGEND)
    else:
        print("Нет результатов: ни по одной категории не набралось данных.")


if __name__ == "__main__":
    main()
```
